# Deps for Colab runtimes.
#
# Setup cell in a Colab notebook:
#     !pip install -q -r requirements-colab.txt && pip install -e . --no-deps
#
# The --no-deps on the editable install is load-bearing: it stops pip from
# re-resolving transformers/torch and burning 5+ minutes per fresh runtime.
# Because --no-deps also skips the *light* deps from pyproject.toml, we
# mirror them here (unpinned) so they get installed by the first pip call.

# Light deps (mirrors pyproject.toml; unpinned so Colab's preinstalled
# versions win when present).
requests
python-dotenv
polars
pydantic
sympy

# Heavy ML deps: minimum bounds only, so Colab's preinstalled torch wins
# the resolve.
transformers>=4.40
accelerate>=0.30
sentence-transformers>=3.0
huggingface-hub>=0.24
faiss-cpu>=1.8

# Hugging Face dataset loader: needed by `scripts/build_math_index.py` and
# `scripts/build_wiki_index.py` to pull the MATH and Wikipedia corpora. tqdm
# and numpy come along transitively via the above so they don't need their
# own line.
datasets>=2.20
# Sparse retrieval companion to FAISS for hybrid Wiki retrieval.
# bm25s uses scipy CSR sparse matrices so an 800k-passage corpus fits in
# ~1.5GB instead of the 15GB+ rank-bm25 needed for the same workload.
bm25s>=0.3.8

# llama-cpp-python: pre-built CUDA 12.2 wheel for Colab T4. The matching
# wheel index is added with --extra-index-url in the bootstrap cell:
#     pip install -r requirements-colab.txt \
#         --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cu122
# Building from source on Colab takes 5+ minutes; the wheel takes seconds.
llama-cpp-python>=0.3
